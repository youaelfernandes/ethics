predictions = {
    "majority": {"approved": 80, "denied": 20},
    "minority": {"approved": 45, "denied": 55}
}

def approval_rate(group):
    total = group["approved"] + group["denied"]
    return group["approved"] / total

maj = approval_rate(predictions["majority"])
min_rate = approval_rate(predictions["minority"])

print(f"Majority approval rate: {maj:.0%}")
print(f"Minority approval rate: {min_rate:.0%}")

ratio = min_rate / maj
print(f"\nDisparate Impact Ratio: {ratio:.2f}")

if ratio < 0.8:
    print("DISPARATE IMPACT - model may be discriminatory")
else:
    print("Passes 80% rule")

import re

records = [
    {"name": "John Doe", "email": "john@mail.com", "phone": "9876543210"},
    {"name": "Alice", "email": "alice@mail.com", "phone": "8765432109"},
    {"name": "Bob", "email": "", "phone": ""},
    {"name": "", "email": "", "ssn": "123-45-6789"}
]

patterns = {
    "Name": r"[A-Z][a-z]+ [A-Z][a-z]+",
    "Email": r"\S+@\S+\.\S+",
    "Phone": r"\b\d{10}\b",
    "SSN": r"\b\d{3}-\d{2}-\d{4}\b"
}

print("Scanning records for PII...\n")

for i, record in enumerate(records, 1):
    for field, value in record.items():
        for pii, pattern in patterns.items():
            if value and re.search(pattern, value):
                print(f"Record {i} - {pii} detected")

print("\nAction: Anonymise or remove PII before training.")

data = [
    {"field": "name", "consent": True, "purpose": "auth", "days": 365},
    {"field": "email", "consent": True, "purpose": "auth", "days": 365},
    {"field": "location_gps", "consent": False, "purpose": "analytics", "days": 730},

    {"field": "browsing_hist", "consent": True, "purpose": "training", "days": 180},

    {"field": "age", "consent": True, "purpose": "training", "days": 90}
]

for d in data:
    status = []

    if not d["consent"]:
        status.append("No consent")

    if d["days"] > 365:

def sentiment(text):
    pos = ["good", "great", "happy", "excellent", "safe", "love"]
    neg = ["bad", "hate", "unsafe", "terrible", "dangerous"]

    text = text.lower()
    score = sum(w in text for w in pos) - sum(w in text for w in neg)

    if score > 0:
        return "POSITIVE"
    if score < 0:
        return "NEGATIVE"
    return "NEUTRAL"


tests = [
    ("This drug is safe", "POSITIVE"),
    ("This drug is SAFE!!!", "POSITIVE"),
    ("This drug is not safe", "NEGATIVE"),
    ("This drug is great", "POSITIVE"),
    ("This drug is terrible", "NEGATIVE")
]

correct = 0

print("Input | Expected | Got | Pass?")

for text, expected in tests:
    got = sentiment(text)
    passed = got == expected

    if passed:
        correct += 1

    print(text, "|", expected, "|", got, "|", passed)

print("Robustness score:", correct, "/", len(tests))

records = [
    ("A", True, False),
    ("A", True, True),
    ("A", False, False),
    ("B", True, False),
    ("B", False, False),
    ("B", False, True)
]

def false_positive_rate(data):
    wrong = sum(1 for r in data if r[1] == False and r[2] == True)
    total = sum(1 for r in data if r[1] == False)
    return wrong / total * 100 if total else 0

groups = {
    "A": [r for r in records if r[0] == "A"],
    "B": [r for r in records if r[0] == "B"]
}

rates = {}

print("FALSE POSITIVE RATE BY GROUP")

for group, data in groups.items():
    rates[group] = false_positive_rate(data)
    print("Group", group, ":", str(int(rates[group])) + "%")

gap = abs(rates["A"] - rates["B"])
print("Gap between groups:", str(int(gap)) + "%")

if gap > 15:
    print("VERDICT: DISPARATE RATES")
else:
    print("VERDICT: comparable rates")

def predict(data):
    weights = {
        "age": 0.1,
        "income": -0.04,
        "debt": 0.5,
        "credit_score": -0.45,
        "employed": -0.3
    }
    return sum(data[k] * weights[k] for k in data)

original = {
    "age": 45,
    "income": 62000,
    "debt": 18000,
    "credit_score": 680,
    "employed": 1
}

base = predict(original)
print("Original prediction:", round(base, 2))

print("\nFeature       Original  Perturbed  Impact")

for feature, value in original.items():
    changed = original.copy()
    changed[feature] = value * 1.1

    impact = predict(changed) - base

    print(feature, value, round(changed[feature], 2), round(impact, 3))

    {"name": "Bob Singh", "age": 52, "email": "bob@mail.com", "diagnosis": "diabetes"},

]

def anonymise(p):
    return {
        "name": "REDACTED",
        "age_band": str(p["age"] // 10 * 10) + "s",
        "email": "REDACTED",
        "diagnosis": p["diagnosis"]
    }

print("BEFORE anonymisation:")
for p in patients:
    print(p)

anon = [anonymise(p) for p in patients]

print("\nAFTER anonymisation:")
for p in anon:
    print(p)

ids = sum(1 for p in anon if p["name"] != "REDACTED" or p["email"] != "REDACTED")

print("\nDirect identifiers remaining:", ids, "/", len(anon))
